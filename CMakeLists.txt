cmake_minimum_required(VERSION 3.21)
project(DecodeForge VERSION 1.0.0 LANGUAGES CXX)

option(DF_ENABLE_CUDA "Build real CUDA kernels (requires nvcc and NVIDIA GPU at runtime)" ON)
option(DF_ENABLE_ASAN "Build with AddressSanitizer/UBSan (CPU verification only)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Source and comments are UTF-8, independent of the Windows system code page.
if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>
                      $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
endif()

add_library(decodeforge_core STATIC src/core.cpp src/ops_cpu.cpp)
target_include_directories(decodeforge_core PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_compile_features(decodeforge_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(decodeforge_core PRIVATE /W4)
else()
  target_compile_options(decodeforge_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

# Preserve user-supplied architecture (e.g. -DCMAKE_CUDA_ARCHITECTURES=86).
# RTX 4060 Laptop and RTX 4090 default to Ada SM 89.
if(DF_ENABLE_CUDA AND NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA SM architecture(s), e.g. 89 or 86")
endif()

if(DF_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12.0 REQUIRED)
  add_library(decodeforge_gpu STATIC src/gpu.cu src/ops_gpu.cu)
  target_link_libraries(decodeforge_gpu PUBLIC decodeforge_core CUDA::cudart CUDA::cublas)
  set_target_properties(decodeforge_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    CUDA_SEPARABLE_COMPILATION OFF)
  message(STATUS "DecodeForge CUDA architectures: ${CMAKE_CUDA_ARCHITECTURES}")
else()
  add_library(decodeforge_gpu STATIC src/gpu_stub.cpp src/ops_stub.cpp)
  target_link_libraries(decodeforge_gpu PUBLIC decodeforge_core)
endif()

target_include_directories(decodeforge_gpu PUBLIC ${PROJECT_SOURCE_DIR}/include)
add_executable(decodeforge_cli src/main.cpp)
target_link_libraries(decodeforge_cli PRIVATE decodeforge_gpu)
add_executable(decodeforge_ops_cli src/ops_cli.cpp)
target_link_libraries(decodeforge_ops_cli PRIVATE decodeforge_gpu)
add_executable(decodeforge_accuracy_cli src/accuracy_cli.cpp)
target_link_libraries(decodeforge_accuracy_cli PRIVATE decodeforge_gpu)

include(CTest)
if(BUILD_TESTING)
  add_executable(decodeforge_cpu_tests tests/cpu_test.cpp)
  target_link_libraries(decodeforge_cpu_tests PRIVATE decodeforge_core)
  add_test(NAME cpu_invariants COMMAND decodeforge_cpu_tests)
  add_executable(decodeforge_ops_cpu_tests tests/ops_cpu_test.cpp)
  target_link_libraries(decodeforge_ops_cpu_tests PRIVATE decodeforge_core)
  add_test(NAME cpu_ops_invariants COMMAND decodeforge_ops_cpu_tests)
  add_test(NAME cli_invalid_numeric COMMAND ${CMAKE_COMMAND}
    -DOPS_CLI=$<TARGET_FILE:decodeforge_ops_cli> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_invalid_test.cmake)
  if(DF_ENABLE_CUDA)
    add_executable(decodeforge_gpu_tests tests/gpu_test.cpp)
    target_link_libraries(decodeforge_gpu_tests PRIVATE decodeforge_gpu)
    add_test(NAME gpu_correctness COMMAND decodeforge_gpu_tests)
    set_tests_properties(gpu_correctness PROPERTIES LABELS "requires_nvidia_gpu" TIMEOUT 180)
    add_executable(decodeforge_ops_gpu_tests tests/ops_gpu_test.cpp)
    target_link_libraries(decodeforge_ops_gpu_tests PRIVATE decodeforge_gpu)
    add_test(NAME gpu_ops_correctness COMMAND decodeforge_ops_gpu_tests)
    set_tests_properties(gpu_ops_correctness PROPERTIES LABELS "requires_nvidia_gpu" TIMEOUT 180)
  else()
    add_executable(decodeforge_gpu_stub_trace_tests tests/gpu_stub_trace_test.cpp)
    target_link_libraries(decodeforge_gpu_stub_trace_tests PRIVATE decodeforge_gpu)
    add_test(NAME gpu_stub_trace COMMAND decodeforge_gpu_stub_trace_tests)
  endif()
endif()

if(DF_ENABLE_ASAN)
  if(DF_ENABLE_CUDA)
    message(FATAL_ERROR "ASan/UBSan verification is intended for -DDF_ENABLE_CUDA=OFF")
  endif()
  if(MSVC)
    message(FATAL_ERROR "Use GCC or Clang for this sanitizer profile")
  endif()
  foreach(target decodeforge_core decodeforge_gpu decodeforge_cli decodeforge_ops_cli decodeforge_accuracy_cli decodeforge_cpu_tests decodeforge_ops_cpu_tests decodeforge_gpu_stub_trace_tests)
    if(TARGET ${target})
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
